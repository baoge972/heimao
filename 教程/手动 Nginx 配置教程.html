优化版脚本内容（可直接复制）
#!/bin/bash
set -e

# 颜色定义（提升可读性）
RED="\033[0;31m"
GREEN="\033[0;32m"
YELLOW="\033[1;33m"
NC="\033[0m"

usage() {
    echo "用法: $0 [选项] 域名 端口"
    echo ""
    echo "选项:"
    echo "  -r, --remove 域名      删除已添加的站点"
    echo "  -h, --help            显示帮助"
    echo ""
    echo "示例:"
    echo "  $0 app.example.com 8080                 # 后端默认 127.0.0.1"
    echo "  $0 --remove app.example.com             # 删除站点"
    exit 0
}

# 删除站点逻辑
if [ "$1" == "--remove" ] || [ "$1" == "-r" ]; then
    DOMAIN="$2"
    if [ -z "$DOMAIN" ]; then
        echo -e "${RED}❌ 请指定要删除的域名${NC}"
        exit 1
    fi
    SAFE_NAME=$(echo "$DOMAIN" | sed "s/\./_/g")
    rm -f "/etc/nginx/conf.d/${SAFE_NAME}.conf"
    certbot delete --cert-name "$DOMAIN" --non-interactive 2>/dev/null || true
    systemctl reload nginx
    echo -e "${GREEN}✅ 已删除站点 ${DOMAIN} 及其证书${NC}"
    exit 0
fi

# 帮助
if [ "$1" == "-h" ] || [ "$1" == "--help" ]; then
    usage
fi

# 参数解析（仅域名+端口，后端固定为 127.0.0.1）
DOMAIN="$1"
PORT="$2"
BACKEND_HOST="127.0.0.1"

# 邮箱保持原样，用于证书通知
EMAIL="admin@baoge123.de5.net"

if [ -z "$DOMAIN" ] || [ -z "$PORT" ]; then
    usage
fi

# 检查依赖并自动安装
if ! command -v nginx &> /dev/null; then
    echo -e "${YELLOW}📦 Nginx 未安装，正在安装...${NC}"
    apt update && apt install nginx -y
fi

if ! command -v certbot &> /dev/null; then
    echo -e "${YELLOW}📦 certbot 未安装，正在安装...${NC}"
    apt update && apt install certbot -y
fi

# 确保目录存在
mkdir -p /etc/nginx/conf.d

# 安全文件名处理（支持泛域名占位符）
SAFE_NAME=$(echo "$DOMAIN" | sed "s/\\*/wildcard/g" | sed "s/\./_/g")
NGINX_CONF="/etc/nginx/conf.d/${SAFE_NAME}.conf"

# 覆盖保护
if [ -f "$NGINX_CONF" ]; then
    echo -e "${YELLOW}⚠️ 配置文件 ${SAFE_NAME}.conf 已存在，是否覆盖？(y/n)${NC}"
    read -r CONFIRM
    if [[ ! "$CONFIRM" =~ ^[Yy]$ ]]; then
        echo -e "${RED}已取消操作${NC}"
        exit 1
    fi
fi

# 核心优化：使用占位符避免变量污染，后续通过 sed 精准替换
cat > "$NGINX_CONF" << "NGINXEOF"
server {
    listen 80;
    server_name DOMAIN_PLACEHOLDER;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name DOMAIN_PLACEHOLDER;

    ssl_certificate /etc/letsencrypt/live/DOMAIN_PLACEHOLDER/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/DOMAIN_PLACEHOLDER/privkey.pem;

    location / {
        proxy_pass http://127.0.0.1:PORT_PLACEHOLDER;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
NGINXEOF

# 替换占位符（仅域名和端口）
sed -i "s/DOMAIN_PLACEHOLDER/${DOMAIN}/g" "$NGINX_CONF"
sed -i "s/PORT_PLACEHOLDER/${PORT}/g" "$NGINX_CONF"

echo -e "${GREEN}▶️ 已生成 Nginx 配置文件: ${NGINX_CONF}${NC}"

# 申请 SSL 证书（使用 webroot 模式，避免停止 Nginx）
WEBROOT_PATH="/var/www/html"
if [ -d "/usr/share/nginx/html" ]; then
    WEBROOT_PATH="/usr/share/nginx/html"
fi
echo -e "${GREEN}🔐 开始申请 SSL 证书 (webroot 模式, 网站根: ${WEBROOT_PATH})${NC}"

# 测试 Nginx 配置语法
if ! nginx -t; then
    echo -e "${RED}❌ Nginx 配置文件语法错误，请检查${NC}"
    rm -f "$NGINX_CONF"
    exit 1
fi

systemctl reload nginx

if ! certbot certonly --webroot -w "$WEBROOT_PATH" -d "$DOMAIN" --non-interactive --agree-tos --email "$EMAIL"; then
    echo -e "${RED}❌ SSL 证书申请失败！移除损坏配置并恢复 Nginx${NC}"
    rm -f "$NGINX_CONF"
    systemctl reload nginx
    exit 1
fi

systemctl reload nginx

echo -e "${GREEN}✅ 站点部署完成！${NC}"
echo -e "${GREEN}🔗 访问地址: https://${DOMAIN}${NC}"
echo -e "${GREEN}➡️ 后端代理: http://127.0.0.1:${PORT}${NC}"

🛠️ 优化版核心改进（解决原版报错）

1️⃣ 占位符 + sed 替换 —— 彻底告别 proxy_set_header 变量为空
原脚本直接使用 heredoc 和 $ 符号，在复制粘贴时容易导致 $host 等变量被提前解析成空值，造成配置生成错误。
新版先写入 DOMAIN_PLACEHOLDER / PORT_PLACEHOLDER，再通过 sed 精准替换，保证配置语法完美。

2️⃣ 固定后端为本地回环地址
代理目标固定为 127.0.0.1:用户指定端口，适用于 Docker 端口映射、本地开发服务、各种本机 Web 应用等绝大多数场景，避免因错误指定外部 IP 导致 502 错误。

3️⃣ 证书申请失败自动回滚
如果 certbot 申请证书出错（例如域名未解析、网络问题），脚本会自动删除生成的配置并重载 Nginx，不会遗留损坏站点导致 502 错误。

4️⃣ 使用 webroot 模式，不再停止 Nginx
老脚本 systemctl stop nginx 会造成其它站点短暂中断，新版采用 --webroot 模式，不影响已有业务。
