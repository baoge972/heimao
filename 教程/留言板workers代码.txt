export default {
  async fetch(request, env) {
    try {
      const url = new URL(request.url);
      const ADMIN_PASSWORD = env.ADMIN_PASSWORD;

      if (!ADMIN_PASSWORD) {
        return new Response("❌ 管理员密码未设置", {
          status: 500,
          headers: { "Content-Type": "text/plain;charset=UTF-8" }
        });
      }

      // ========== 工具函数 ==========
      function generateToken() {
        return Date.now().toString(36) + Math.random().toString(36).substr(2, 10);
      }

      function escapeHtml(text) {
        if (!text) return '';
        return text
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#039;');
      }

      function getSession(request) {
        const cookie = request.headers.get("Cookie") || "";
        const match = cookie.match(/session_token=([^;]+)/);
        return match ? match[1] : null;
      }

      async function getCurrentUser(blogId, request, env) {
        const token = getSession(request);
        if (!token) return null;
        const sessionData = await env.FORUM.get(`session_${blogId}_${token}`, "json");
        if (!sessionData) return null;
        if (Date.now() - sessionData.createdAt > 3600000) {
          await env.FORUM.delete(`session_${blogId}_${token}`);
          return null;
        }
        return sessionData;
      }

      // ========== 注册 ==========
      if (request.method === "POST" && url.pathname === "/api/register") {
        try {
          const body = await request.json();
          const { username, password } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!username || !password) {
            return new Response(JSON.stringify({ error: "用户名和密码不能为空" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const userKey = `user_${blogId}_${username}`;
          const usersKey = `users_${blogId}`;

          const existing = await env.FORUM.get(userKey);
          if (existing) {
            return new Response(JSON.stringify({ error: "用户名已存在" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          await env.FORUM.put(userKey, JSON.stringify({
            username,
            password,
            role: "user",
            createdAt: Date.now()
          }));

          const users = await env.FORUM.get(usersKey, "json") || [];
          users.push(username);
          await env.FORUM.put(usersKey, JSON.stringify(users));

          return new Response(JSON.stringify({ success: true, message: "注册成功" }), {
            status: 200,
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "注册失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 用户登录 ==========
      if (request.method === "POST" && url.pathname === "/api/login") {
        try {
          const body = await request.json();
          const { username, password } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!username || !password) {
            return new Response(JSON.stringify({ error: "用户名和密码不能为空" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const userKey = `user_${blogId}_${username}`;
          const userData = await env.FORUM.get(userKey, "json");

          if (!userData) {
            return new Response(JSON.stringify({ error: "用户不存在" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (userData.password !== password) {
            return new Response(JSON.stringify({ error: "密码错误" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          // 删除该用户所有旧会话
          const list = await env.FORUM.list({ prefix: `session_${blogId}_` });
          for (const key of list.keys) {
            const data = await env.FORUM.get(key.name, "json");
            if (data && data.username === username) {
              await env.FORUM.delete(key.name);
            }
          }

          const token = generateToken();
          await env.FORUM.put(`session_${blogId}_${token}`, JSON.stringify({
            username,
            role: userData.role || "user",
            createdAt: Date.now()
          }), { expirationTtl: 3600 });

          return new Response(JSON.stringify({ success: true, token, username }), {
            headers: {
              "Content-Type": "application/json",
              "Set-Cookie": `session_token=${token}; Path=/; Max-Age=3600; SameSite=Lax`
            }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "登录失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 发表留言 ==========
      if (request.method === "POST" && url.pathname === "/api/post") {
        try {
          const body = await request.json();
          const { content } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const user = await getCurrentUser(blogId, request, env);
          if (!user) {
            return new Response(JSON.stringify({ error: "请先登录" }), {
              status: 401,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!content || content.trim().length === 0) {
            return new Response(JSON.stringify({ error: "请输入内容" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const msgKey = `messages_${blogId}_${user.username}`;
          const messages = await env.FORUM.get(msgKey, "json") || [];
          messages.push({
            id: generateToken(),
            name: user.username,
            content: content.trim(),
            time: new Date().toLocaleString('zh-CN'),
            replies: []
          });
          await env.FORUM.put(msgKey, JSON.stringify(messages));

          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "发表失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 删除自己的留言 ==========
      if (request.method === "POST" && url.pathname === "/api/delete") {
        try {
          const body = await request.json();
          const { id } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const user = await getCurrentUser(blogId, request, env);
          if (!user) {
            return new Response(JSON.stringify({ error: "请先登录" }), {
              status: 401,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!id) {
            return new Response(JSON.stringify({ error: "缺少留言ID" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const msgKey = `messages_${blogId}_${user.username}`;
          const messages = await env.FORUM.get(msgKey, "json") || [];
          const idx = messages.findIndex(m => m.id === id);

          if (idx === -1) {
            return new Response(JSON.stringify({ error: "留言不存在" }), {
              status: 404,
              headers: { "Content-Type": "application/json" }
            });
          }

          messages.splice(idx, 1);
          await env.FORUM.put(msgKey, JSON.stringify(messages));

          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "删除失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 管理员登录 ==========
      if (request.method === "POST" && url.pathname === "/admin/login") {
        try {
          const body = await request.json();
          const { password } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!password) {
            return new Response(JSON.stringify({ error: "请输入密码" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (password !== ADMIN_PASSWORD) {
            return new Response(JSON.stringify({ error: "密码错误" }), {
              status: 403,
              headers: { "Content-Type": "application/json" }
            });
          }

          // 删除所有旧会话
          const list = await env.FORUM.list({ prefix: `session_${blogId}_` });
          for (const key of list.keys) {
            await env.FORUM.delete(key.name);
          }

          const token = generateToken();
          await env.FORUM.put(`session_${blogId}_${token}`, JSON.stringify({
            username: "admin",
            role: "admin",
            createdAt: Date.now()
          }), { expirationTtl: 3600 });

          return new Response(JSON.stringify({ success: true }), {
            headers: {
              "Content-Type": "application/json",
              "Set-Cookie": `session_token=${token}; Path=/; Max-Age=3600; SameSite=Lax`
            }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "登录失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 管理员删除留言 ==========
      if (request.method === "POST" && url.pathname === "/api/admin/delete") {
        try {
          const body = await request.json();
          const { id } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const user = await getCurrentUser(blogId, request, env);
          if (!user || user.role !== "admin") {
            return new Response(JSON.stringify({ error: "无权限" }), {
              status: 403,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!id) {
            return new Response(JSON.stringify({ error: "缺少留言ID" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const users = await env.FORUM.get(`users_${blogId}`, "json") || [];
          let found = false;

          for (const u of users) {
            const msgKey = `messages_${blogId}_${u}`;
            const messages = await env.FORUM.get(msgKey, "json") || [];
            const idx = messages.findIndex(m => m.id === id);

            if (idx !== -1) {
              messages.splice(idx, 1);
              await env.FORUM.put(msgKey, JSON.stringify(messages));
              found = true;
              break;
            }
          }

          if (!found) {
            return new Response(JSON.stringify({ error: "留言不存在" }), {
              status: 404,
              headers: { "Content-Type": "application/json" }
            });
          }

          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "删除失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 管理员删除回复 ==========
      if (request.method === "POST" && url.pathname === "/api/admin/deleteReply") {
        try {
          const body = await request.json();
          const { messageId, replyIndex } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const user = await getCurrentUser(blogId, request, env);
          if (!user || user.role !== "admin") {
            return new Response(JSON.stringify({ error: "无权限" }), {
              status: 403,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!messageId || replyIndex === undefined) {
            return new Response(JSON.stringify({ error: "缺少参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const users = await env.FORUM.get(`users_${blogId}`, "json") || [];
          let found = false;

          for (const u of users) {
            const msgKey = `messages_${blogId}_${u}`;
            const messages = await env.FORUM.get(msgKey, "json") || [];
            const idx = messages.findIndex(m => m.id === messageId);

            if (idx !== -1) {
              if (messages[idx].replies && messages[idx].replies.length > replyIndex) {
                messages[idx].replies.splice(replyIndex, 1);
                await env.FORUM.put(msgKey, JSON.stringify(messages));
                found = true;
                break;
              }
            }
          }

          if (!found) {
            return new Response(JSON.stringify({ error: "留言或回复不存在" }), {
              status: 404,
              headers: { "Content-Type": "application/json" }
            });
          }

          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "删除回复失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 管理员回复 ==========
      if (request.method === "POST" && url.pathname === "/api/admin/reply") {
        try {
          const body = await request.json();
          const { messageId, content } = body;
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const user = await getCurrentUser(blogId, request, env);
          if (!user || user.role !== "admin") {
            return new Response(JSON.stringify({ error: "无权限" }), {
              status: 403,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!messageId) {
            return new Response(JSON.stringify({ error: "缺少留言ID" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          if (!content || content.trim().length === 0) {
            return new Response(JSON.stringify({ error: "请输入回复内容" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const users = await env.FORUM.get(`users_${blogId}`, "json") || [];
          let found = false;

          for (const u of users) {
            const msgKey = `messages_${blogId}_${u}`;
            const messages = await env.FORUM.get(msgKey, "json") || [];
            const idx = messages.findIndex(m => m.id === messageId);

            if (idx !== -1) {
              if (!messages[idx].replies) messages[idx].replies = [];
              messages[idx].replies.push({
                name: "管理员",
                content: content.trim(),
                time: new Date().toLocaleString('zh-CN')
              });
              await env.FORUM.put(msgKey, JSON.stringify(messages));
              found = true;
              break;
            }
          }

          if (!found) {
            return new Response(JSON.stringify({ error: "留言不存在" }), {
              status: 404,
              headers: { "Content-Type": "application/json" }
            });
          }

          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json" }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "回复失败: " + e.message }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 退出登录 ==========
      if (request.method === "POST" && url.pathname === "/api/logout") {
        try {
          const body = await request.json();
          const blogId = body.blog || url.searchParams.get("blog");
          if (!blogId) {
            return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
              status: 400,
              headers: { "Content-Type": "application/json" }
            });
          }

          const token = getSession(request);
          if (token) {
            await env.FORUM.delete(`session_${blogId}_${token}`);
          }
          return new Response(JSON.stringify({ success: true }), {
            headers: {
              "Content-Type": "application/json",
              "Set-Cookie": "session_token=; Path=/; Max-Age=0"
            }
          });
        } catch (e) {
          return new Response(JSON.stringify({ error: "退出失败" }), {
            status: 500,
            headers: { "Content-Type": "application/json" }
          });
        }
      }

      // ========== 统计接口 ==========
      if (request.method === "GET" && url.pathname === "/api/count") {
        const blogId = url.searchParams.get("blog");
        if (!blogId) {
          return new Response(JSON.stringify({ error: "缺少 blog 参数" }), {
            status: 400,
            headers: { "Content-Type": "application/json;charset=UTF-8" }
          });
        }

        const users = await env.FORUM.get(`users_${blogId}`, "json") || [];
        let total = 0;
        for (const u of users) {
          const msgs = await env.FORUM.get(`messages_${blogId}_${u}`, "json") || [];
          total += msgs.length;
        }
        return new Response(JSON.stringify({ count: total }), {
          headers: { "Content-Type": "application/json;charset=UTF-8" }
        });
      }

      // ========== 留言板首页 ==========
      if (request.method === "GET" && url.pathname === "/") {
        const blogId = url.searchParams.get("blog");
        if (!blogId) {
          return new Response("❌ 缺少 blog 参数", {
            status: 400,
            headers: { "Content-Type": "text/plain;charset=UTF-8" }
          });
        }

        // 检查登录状态
        const user = await getCurrentUser(blogId, request, env);

        // 获取留言总数
        const users = await env.FORUM.get(`users_${blogId}`, "json") || [];
        let totalCount = 0;
        for (const u of users) {
          const msgs = await env.FORUM.get(`messages_${blogId}_${u}`, "json") || [];
          totalCount += msgs.length;
        }

        // 如果已登录（普通用户），显示用户自己的留言
        let messagesHtml = '';
        let isLoggedIn = false;
        let username = '';
        let isAdmin = false;
        if (user) {
          isLoggedIn = true;
          username = user.username;
          isAdmin = user.role === "admin";
          
          // 如果是管理员，不显示留言列表
          if (!isAdmin) {
            const messages = await env.FORUM.get(`messages_${blogId}_${username}`, "json") || [];
            if (messages.length === 0) {
              messagesHtml = '<div class="empty">✨ 暂无留言，发表第一条吧！</div>';
            } else {
              for (const m of messages) {
                const repliesHtml = (m.replies && m.replies.length > 0) ? 
                  `<div class="replies">${m.replies.map(r => `
                    <div class="reply-item">
                      <div class="reply-meta">
                        <span class="reply-name">📌 ${escapeHtml(r.name)}</span>
                        <span class="reply-time">${escapeHtml(r.time || '')}</span>
                      </div>
                      <div class="reply-content">${escapeHtml(r.content)}</div>
                    </div>
                  `).join('')}</div>` : '';
                
                messagesHtml += `
                  <div class="message-item" data-id="${escapeHtml(m.id)}">
                    <div class="meta">
                      <span class="name">${escapeHtml(m.name)}</span>
                      <span class="time">${escapeHtml(m.time || '')}</span>
                    </div>
                    <div class="content">${escapeHtml(m.content)}</div>
                    <div class="actions">
                      <button class="delete-btn" onclick="deleteMessage('${escapeHtml(m.id)}')">删除</button>
                    </div>
                    ${repliesHtml}
                  </div>
                `;
              }
            }
          }
        }

        const html = `
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=yes">
    <title>📝 留言板</title>
    <style>
        * { margin: 0; padding: 0; box-sizing: border-box; }
        html, body { 
          height: 100%; 
          font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Arial, sans-serif; 
          background: #f0f2f5; 
          color: #1a1a2e; 
          font-size: 16px; 
          overflow: hidden; 
          padding: 16px;
        }
        .app { 
          display: flex; 
          flex-direction: column; 
          height: 100%; 
          max-width: 720px; 
          margin: 0 auto; 
          background: #ffffff; 
          border-radius: 16px; 
          box-shadow: 0 4px 30px rgba(0,0,0,0.08); 
          overflow: hidden;
        }
        .header { display: flex; justify-content: space-between; align-items: center; padding: 14px 18px; background: #ffffff; border-bottom: 1px solid #e8ecf1; flex-shrink: 0; flex-wrap: wrap; gap: 8px; }
        .header h1 { font-size: 20px; font-weight: 700; }
        .header h1 span { color: #4CAF50; }
        .header .close-btn { background: #ff4444; color: white; border: none; padding: 6px 16px; border-radius: 20px; cursor: pointer; font-size: 13px; transition: background 0.2s; }
        .header .close-btn:hover { background: #cc0000; }
        .header .user-info { display: flex; align-items: center; gap: 10px; flex-wrap: wrap; }
        .header .user-info .username { font-size: 14px; color: #555; }
        .header .logout-btn { background: #888; color: white; border: none; padding: 6px 16px; border-radius: 20px; font-size: 13px; cursor: pointer; transition: background 0.2s; }
        .header .logout-btn:hover { background: #666; }
        .header .count-badge { background: #f0f2f5; padding: 4px 14px; border-radius: 20px; font-size: 13px; color: #555; }

        .login-prompt { display: flex; flex-direction: column; align-items: center; justify-content: center; flex: 1; padding: 40px 20px; text-align: center; }
        .login-prompt .icon { font-size: 48px; margin-bottom: 16px; }
        .login-prompt .title { font-size: 20px; font-weight: 600; margin-bottom: 8px; }
        .login-prompt .desc { color: #888; font-size: 14px; margin-bottom: 20px; }
        .login-prompt .btn-group { display: flex; gap: 12px; flex-wrap: wrap; justify-content: center; }
        .login-prompt .btn-group .login-btn { padding: 10px 32px; background: #4CAF50; color: white; border: none; border-radius: 8px; font-size: 16px; font-weight: 600; cursor: pointer; text-decoration: none; transition: background 0.2s; }

<script>
    const BLOG_ID = "${escapeHtml(blogId)}";
    let toastTimer = null;

    function showToast(msg, type = '') {
        const toast = document.getElementById('toast');
        toast.textContent = msg;
        toast.className = 'toast ' + type;
        clearTimeout(toastTimer);
        void toast.offsetWidth;
        toast.classList.add('show');
        toastTimer = setTimeout(() => { toast.classList.remove('show'); }, 2500);
    }

    window.adminDelete = async function(id) {
        if (!confirm('确定要删除这条留言吗？')) return;
        try {
            const resp = await fetch('/api/admin/delete', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ blog: BLOG_ID, id })
            });
            const result = await resp.json();
            if (resp.ok) {
                showToast('✅ 删除成功', 'success');
                setTimeout(() => location.reload(), 500);
            } else {
                showToast('❌ ' + (result.error || '删除失败'), 'error');
            }
        } catch (e) {
            showToast('❌ 请求失败', 'error');
        }
    };

    window.deleteReply = async function(messageId, replyIndex) {
        if (!confirm('确定要删除这条回复吗？')) return;
        try {
            const resp = await fetch('/api/admin/deleteReply', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ blog: BLOG_ID, messageId, replyIndex })
            });
            const result = await resp.json();
            if (resp.ok) {
                showToast('✅ 删除成功', 'success');
                setTimeout(() => location.reload(), 500);
            } else {
                showToast('❌ ' + (result.error || '删除失败'), 'error');
            }
        } catch (e) {
            showToast('❌ 请求失败', 'error');
        }
    };

    window.toggleReply = function(id) {
        const form = document.getElementById('reply-form-' + id);
        if (form.style.display === 'none' || !form.style.display) {
            form.style.display = 'block';
            document.getElementById('reply-content-' + id).focus();
        } else {
            form.style.display = 'none';
        }
    };

    window.submitReply = async function(id) {
        const textarea = document.getElementById('reply-content-' + id);
        const content = textarea.value.trim();
        if (!content) {
            showToast('请输入回复内容', 'error');
            return;
        }

        try {
            const resp = await fetch('/api/admin/reply', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ blog: BLOG_ID, messageId: id, content })
            });
            const result = await resp.json();
            if (resp.ok) {
                showToast('✅ 回复成功', 'success');
                setTimeout(() => location.reload(), 500);
            } else {
                showToast('❌ ' + (result.error || '回复失败'), 'error');
            }
        } catch (e) {
            showToast('❌ 请求失败', 'error');
        }
    };

    document.querySelectorAll('.reply-form textarea').forEach(el => {
        el.addEventListener('keydown', function(e) {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                const id = this.id.replace('reply-content-', '');
                submitReply(id);
            }
        });
    });
</script>
