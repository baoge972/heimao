// 清理过期的频率限制记录（在请求处理中执行）
function cleanRateLimitStore() {
    const now = Date.now();
    for (const [key, time] of rateLimitStore.entries()) {
        if (now - time > RATE_LIMIT.windowMs) {
            rateLimitStore.delete(key);
        }
    }
}

// 计算HTML的ETag（基于内容长度）
function generateETag(html) {
    // 使用内容长度和简单哈希
    let hash = 0;
    for (let i = 0; i < html.length; i++) {
        hash = ((hash << 5) - hash) + html.charCodeAt(i);
        hash = hash & hash;
    }
    return `"${html.length}-${Math.abs(hash)}"`;
}

// 频率限制检查
function checkRateLimit(request) {
    const ip = request.headers.get('CF-Connecting-IP') || 
               request.headers.get('X-Forwarded-For')?.split(',')[0] || 
               'unknown';
    const now = Date.now();
    
    // 清理过期记录
    cleanRateLimitStore();
    
    // 统计IP在时间窗口内的请求数
    let requestCount = 0;
    for (const [key, time] of rateLimitStore.entries()) {
        if (key.startsWith(ip) && now - time < RATE_LIMIT.windowMs) {
            requestCount++;
        }
    }
    
    if (requestCount >= RATE_LIMIT.maxRequests) {
        return false;
    }
    
    // 记录本次请求
    rateLimitStore.set(`${ip}-${now}`, now);
    return true;
}

// 检查是否是爬虫
function isBot(request) {
    const userAgent = request.headers.get('User-Agent')?.toLowerCase() || '';
    const bots = ['bot', 'crawler', 'spider', 'scanner', 'baidu', 'googlebot', 'bingbot', 'slurp', 'duckduckbot', 'facebookexternalhit', 'twitterbot'];
    return bots.some(bot => userAgent.includes(bot));
}

// 处理条件请求（304 Not Modified）
function handleConditionalRequest(request, etag) {
    const ifNoneMatch = request.headers.get('If-None-Match');
    
    if (ifNoneMatch === etag) {
        return true;
    }
    
    return false;
}

// 主请求处理函数
async function handleRequest(request) {
    const url = new URL(request.url);
    const method = request.method;
    
    // 1. 健康检查端点（用于监控和负载均衡）
    if (url.pathname === '/health' || url.pathname === '/ping') {
        return new Response('OK', { 
            status: 200,
            headers: {
                'Content-Type': 'text/plain',
                'Cache-Control': 'no-cache',
            }
        });
    }
    
    // 2. 统计端点（可选，用于查看访问统计）
    if (url.pathname === '/stats' && method === 'GET') {
        return new Response(JSON.stringify({
            rateLimitStoreSize: rateLimitStore.size,
            timestamp: Date.now(),
            uptime: performance.now(),
        }), {
            headers: {
                'Content-Type': 'application/json',
                'Cache-Control': 'no-cache',
            }
        });
    }
    
    // 3. OPTIONS请求处理（CORS预检）
    if (method === 'OPTIONS') {
        return new Response(null, {
            status: 204,
            headers: {
                'Access-Control-Allow-Origin': '*',
                'Access-Control-Allow-Methods': 'GET, OPTIONS',
                'Access-Control-Max-Age': '86400',
            }
        });
    }
    
    // 4. 只允许GET请求访问页面
    if (method !== 'GET') {
        return new Response('Method Not Allowed', { status: 405 });
    }
    
    // 5. 爬虫友好处理：给爬虫返回简化版本或延迟响应
    const isBotRequest = isBot(request);
    if (isBotRequest) {
        // 爬虫返回缓存头，减少请求频率
        return new Response(HTML_TEMPLATE, {
            status: 200,
            headers: {
                'Content-Type': 'text/html;charset=UTF-8',
                'Cache-Control': 'public, max-age=86400', // 爬虫缓存24小时
                'X-Robots-Tag': 'index, follow',
            }
        });
    }
    
    // 6. 频率限制检查
    if (!checkRateLimit(request)) {
        return new Response('Too Many Requests - Please try again later.', { 
            status: 429,
            headers: {
                'Content-Type': 'text/plain;charset=UTF-8',
                'Retry-After': '60',
                'Cache-Control': 'no-cache',
            }
        });
    }
    
    // 7. 生成ETag
    const etag = generateETag(HTML_TEMPLATE);
    
    // 8. 检查条件请求
    if (handleConditionalRequest(request, etag)) {
        return new Response(null, { 
            status: 304,
            headers: {
                'ETag': etag,
                'Cache-Control': 'public, max-age=3600',
                'Cache-Tag': 'domain-monitor,html',
            }
        });
    }
    
    // 9. 返回完整响应（带完整的缓存头）
    return new Response(HTML_TEMPLATE, {
        status: 200,
        headers: {
            'Content-Type': 'text/html;charset=UTF-8',
            
            // 缓存控制（关键优化）
            'Cache-Control': 'public, max-age=3600, s-maxage=7200, stale-while-revalidate=300',
            'CDN-Cache-Control': 'public, max-age=86400',
            'Cloudflare-CDN-Cache-Control': 'public, max-age=86400',
            
            // 缓存验证
            'ETag': etag,
            
            // 安全头
            'X-Content-Type-Options': 'nosniff',
            'X-Frame-Options': 'SAMEORIGIN',
            'X-XSS-Protection': '1; mode=block',
            'Referrer-Policy': 'strict-origin-when-cross-origin',
            
            // 缓存标记（用于CDN缓存清除）
            'Cache-Tag': 'domain-monitor,html',
            
            // 变化头
            'Vary': 'Accept-Encoding, User-Agent',
        },
    });
}

// Worker入口
addEventListener('fetch', event => {
