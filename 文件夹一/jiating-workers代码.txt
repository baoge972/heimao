    <script>
        const { createApp, ref, computed, onMounted } = Vue;

            setup() {

                const isMobile = window.innerWidth <= 480;

                // ===== 备忘录独立密码 =====
                const showMemoAuth = ref(false);

                        memoAuthed.value = true;

                    const nextIndex = (currentIndex + 1) % themes.length;
                    currentTheme.value = themes[nextIndex];
                    document.documentElement.setAttribute('data-theme', currentTheme.value);
                    localStorage.setItem('dashboard_theme', currentTheme.value);
                };

                    const saved = localStorage.getItem('dashboard_theme');
                    if (saved && ['dark', 'light', 'blue', 'green'].includes(saved)) {

                        document.documentElement.setAttribute('data-theme', saved);
                    }

                // ===== 记账 =====
                const API_BASE = 'https://shujv.baoge.wang';
                const ACCOUNTING_TOKEN_KEY = 'dashboard_accounting_token';

                const showAccountingLogin = ref(false);
                const accountingLoading = ref(false);

                const accountingLoginForm = ref({ username: '', password: '' });
                const accountingToken = ref('');

                        const isRed = function(type) {

                            return lowerType.includes('红') || lowerType.includes('喜') || lowerType.includes('结婚') ||

                        const isWhite = function(type) {

                            const lowerType = type.toLowerCase();
                            return lowerType.includes('白') || lowerType.includes('丧') || lowerType.includes('去世') ||
                                lowerType.includes('悼') || lowerType.includes('祭') || lowerType.includes('葬礼');
                        };

                        const totalEvents = events.length + uniqueAccountBookEvents.length;

                        let redEvents = 0;
                        events.forEach(function(e) {
                            if (isRed(e.type)) redEvents++;
                        });
                        uniqueAccountBookEvents.forEach(function(e) {
                            if (isRed(e.type)) redEvents++;
                        });

                        let whiteEvents = 0;
                        events.forEach(function(e) {
                            if (isWhite(e.type)) whiteEvents++;

                            receiveAmount: Math.round(receiveAmount),
