const adminTokens = new Set();

const server = http.createServer((req, res) => {
  const parsedUrl = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  const pathname = parsedUrl.pathname;

  res.setHeader('Access-Control-Allow-Origin', '*');

  if (req.method === 'POST' && pathname === '/api/login') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      try {
        const { password } = JSON.parse(body);
        if (password === ADMIN_PASSWORD) {
          const token = Math.random().toString(36).substring(2) + Date.now().toString(36);
          adminTokens.add(token);
          res.writeHead(200, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ success: true, token }));
        } else {
          res.writeHead(401, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ success: false }));
        }
      } catch (e) { res.writeHead(400); res.end('Bad Request'); }
    });
    return;
  }

  if (req.method === 'GET' && pathname === '/dashboard') {
    if (parsedUrl.searchParams.get('token') && adminTokens.has(parsedUrl.searchParams.get('token'))) {
      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(DASHBOARD_PAGE);
    } else {
      res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(NOT_FOUND_PAGE);
    }
    return;
  }

  if (req.method === 'POST' && pathname === '/api/create') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      try {
        const { content, maxViews, expireHours } = JSON.parse(body);
        const id = generateId();
        vault[id] = { content, viewsLeft: maxViews || 1, expireAt: Date.now() + (expireHours || 24) * 3600 * 1000 };
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ success: true, id }));
      } catch (e) { res.writeHead(400); res.end('Bad Request'); }
    });
    return;
  }

  if (req.method === 'GET' && (pathname === '/' || pathname === '/index.html')) {
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
    res.end(LOGIN_PAGE);
    return;
  }

  if (req.method === 'GET' && pathname.length >= 2 && !pathname.startsWith('/api/') && pathname !== '/dashboard' && pathname !== '/404') {
    const id = pathname.substring(1);
    const item = vault[id];

    if (!item) { res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(NOT_FOUND_PAGE); return; }
    if (Date.now() > item.expireAt) { delete vault[id]; res.writeHead(410, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(NOT_FOUND_PAGE); return; }

    item.viewsLeft -= 1;
    if (item.viewsLeft <= 0) delete vault[id];

    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
    res.end(VIEW_PAGE(item.content, Math.max(0, item.viewsLeft)));
    return;
  }

  res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
  res.end(NOT_FOUND_PAGE);
});

// 🚀 原生下载器 (无需 axios)
// ==========================================
function downloadFile(url, destPath) {
  return new Promise((resolve, reject) => {
    const file = fs.createWriteStream(destPath);
    https.get(url, (response) => {
      // 处理重定向
      if (response.statusCode >= 300 && response.statusCode < 400 && response.headers.location) {
        file.close();
        fs.unlink(destPath, () => {});
        return downloadFile(response.headers.location, destPath).then(resolve).catch(reject);
      }
      if (response.statusCode !== 200) {
        file.close();
        fs.unlink(destPath, () => {});
        return reject(new Error(`Download failed with status code ${response.statusCode}`));
      }
      response.pipe(file);
      file.on('finish', () => { file.close(); resolve(); });
    }).on('error', (err) => {
      file.close();
      fs.unlink(destPath, () => {});
      reject(err);
    });
  });
}

async function runTunnel() {
  if (!ARGO_AUTH || !ARGO_DOMAIN) {
    console.log('\n⚠️ 警告: 你没有填写 ARGO_AUTH 或 ARGO_DOMAIN！隧道将无法启动。');
    console.log('请在上方配置区域填入你的 Token 和域名，然后重启。\n');
    return;
  }

  if (!fs.existsSync(botPath)) {
    console.log(`[系统] 开始下载 cloudflared 隧道组件...`);
    const arch = os.arch();
    const fileUrl = (arch === 'arm' || arch === 'arm64') 
      ? "https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-arm64"
      : "https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64";
    
    try {
      await downloadFile(fileUrl, botPath);
      fs.chmodSync(botPath, 0o775);
      console.log('[系统] ✅ cloudflared 下载并授权成功');
    } catch (err) {
      console.error('[系统] ❌ 下载 cloudflared 失败:', err.message);
      return;
    }
  }

  console.log(`[隧道] 正在启动... 目标域名: https://${ARGO_DOMAIN}`);
  
  const child = spawn(botPath, [
    'tunnel', '--edge-ip-version', 'auto', '--no-autoupdate', 
    '--protocol', 'http2', 'run', '--token', ARGO_AUTH
  ]);

  child.stdout.on('data', (data) => console.log(`[Cloudflare] ${data.toString().trim()}`));
  child.stderr.on('data', (data) => console.error(`[Cloudflare] ${data.toString().trim()}`));
  child.on('close', (code) => console.log(`[隧道] 进程已退出，代码: ${code}`));
}

// 启动服务器
server.listen(PORT, async () => {
  console.log(`[系统] ✅ 本地服务已启动，端口: ${PORT}`);
  await runTunnel();
});
