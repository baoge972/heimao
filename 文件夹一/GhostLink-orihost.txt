const NOT_FOUND_PAGE = `<!DOCTYPE html><html lang="zh-CN"><head><meta charset="UTF-8"><title>404 Not Found</title><link rel="icon" type="image/svg+xml" href="${FAVICON_URL}"></head><body style="background:#0a0a0a;color:#333;display:flex;align-items:center;justify-content:center;height:100vh;font-family:sans-serif;margin:0;"><h1 style="font-size:5rem;margin:0;">404</h1></body></html>`;

// 记录登录的 token
const adminTokens = new Set();

const server = http.createServer((req, res) => {
  const parsedUrl = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  const pathname = parsedUrl.pathname;
  
  res.setHeader('Access-Control-Allow-Origin', '*');

  // 1. 登录验证 API
  if (req.method === 'POST' && pathname === '/api/login') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      try {
        const { password } = JSON.parse(body);
        if (password === ADMIN_PASSWORD) {
          const token = Math.random().toString(36).substring(2) + Date.now().toString(36);
          adminTokens.add(token);
          res.writeHead(200, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ success: true, token }));
        } else {
          res.writeHead(401, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ success: false }));
        }
      } catch (err) {
        res.writeHead(400); res.end('Bad Request');
      }
    });
    return;
  }

  // 2. 仪表盘页面
  if (req.method === 'GET' && pathname === '/dashboard') {
    const token = parsedUrl.searchParams.get('token');
    if (token && adminTokens.has(token)) {
      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(DASHBOARD_PAGE);
    } else {
      res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(NOT_FOUND_PAGE);
    }
    return;
  }

  // 3. 生成 API
  if (req.method === 'POST' && pathname === '/api/create') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      try {
        const { content, maxViews, expireHours } = JSON.parse(body);
        const id = generateId();
        vault[id] = { content, viewsLeft: maxViews || 1, expireAt: Date.now() + (expireHours || 24) * 3600 * 1000 };
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ success: true, id }));
      } catch (err) {
        res.writeHead(400); res.end('Bad Request');
      }
    });
    return;
  }

  // 4. 根路径
  if (req.method === 'GET' && (pathname === '/' || pathname === '/index.html')) {
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
    res.end(LOGIN_PAGE);
    return;
  }

  // 5. 阅后即焚访问路径
  if (req.method === 'GET' && pathname.length >= 2 && !pathname.startsWith('/api/') && pathname !== '/dashboard' && pathname !== '/404') {
    const id = pathname.substring(1); 
    const item = vault[id];

    if (!item) {
      res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(NOT_FOUND_PAGE);
      return;
    }

    if (Date.now() > item.expireAt) {
      delete vault[id];
      res.writeHead(410, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end(NOT_FOUND_PAGE);
      return;
    }

    item.viewsLeft -= 1;
    if (item.viewsLeft <= 0) {
      delete vault[id]; 
    }

    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
    res.end(VIEW_PAGE(item.content, Math.max(0, item.viewsLeft)));
    return;
  }

  // 6. 兜底 404
  res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
  res.end(NOT_FOUND_PAGE);
});

setInterval(() => {
  const now = Date.now();
  for (const id in vault) {
    if (now > vault[id].expireAt) {
      delete vault[id];
    }
  }
}, 10 * 60 * 1000);

server.listen(PORT, () => {
  console.log('GhostLink 安全运行在端口 ' + PORT);
  console.log('后台密码: ' + ADMIN_PASSWORD);
  console.log('访问主页: http://你的IP:' + PORT + '/');
});
