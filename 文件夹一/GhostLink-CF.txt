// Cloudflare Workers 专属纯净版 (终极 Logo 修复版)
// 环境变量配置要求：
// 1. Variables -> ADMIN_PASSWORD (你的后台密码)
// 2. KV Namespace Bindings -> GHOST_KV (你的KV命名空间)

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const pathname = url.pathname;
    
    const ADMIN_PASSWORD = env.ADMIN_PASSWORD || "8888"; 
    
    // 🌟 终极防兼容性写法：直接内联未经编码的 SVG (仅对引号做转义)
    const FAVICON_URL = "data:image/svg+xml,%3Csvg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 24 24' fill='none' stroke='%23e11d48' stroke-width='2' stroke-linecap='round' stroke-linejoin='round'%3E%3Cpath d='M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z'/%3E%3Cpath d='M9 10h.01'/%3E%3Cpath d='M15 10h.01'/%3E%3Cpath d='M12 14c-1.5 0-2.5-1-2.5-1s1 2.5 2.5 2.5 2.5-2.5 2.5-2.5-1 1-2.5 1z'/%3E%3C/svg%3E";

    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET,HEAD,POST,OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    // 1. 登录页面
    if (request.method === "GET" && (pathname === "/" || pathname === "/index.html")) {
      return new Response(getLoginPage(FAVICON_URL), {
        headers: { "Content-Type": "text/html; charset=utf-8", ...corsHeaders },
      });
    }

    // 2. 登录验证 API
    if (request.method === "POST" && pathname === "/api/login") {
      try {
        const body = await request.json();
        if (body.password === ADMIN_PASSWORD) {
          return new Response(JSON.stringify({ success: true }), {
            headers: { "Content-Type": "application/json", ...corsHeaders },
          });
        }
      } catch (e) {}
      return new Response(JSON.stringify({ success: false }), {
        status: 401,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    // 3. 控制台页面
    if (request.method === "GET" && pathname === "/dashboard") {
      return new Response(getDashboardPage(FAVICON_URL), {
        headers: { "Content-Type": "text/html; charset=utf-8", ...corsHeaders },
      });
    }

    // 4. 创建 API
    if (request.method === "POST" && pathname === "/api/create") {
      try {
        const { content, expireHours } = await request.json();
        const id = Math.random().toString(36).substring(2, 8);
        
        const ttl = Math.max(60, (expireHours || 24) * 3600); 
        await env.GHOST_KV.put(id, content, { expirationTtl: ttl });

        return new Response(JSON.stringify({ success: true, id }), {
          headers: { "Content-Type": "application/json", ...corsHeaders },
        });
      } catch (e) {
        return new Response(JSON.stringify({ success: false, error: e.message }), {
          status: 400,
          headers: { "Content-Type": "application/json", ...corsHeaders },
        });
      }
    }

    // 5. 阅后即焚查看页
    if (request.method === "GET" && pathname.length >= 2 && !pathname.startsWith("/api/") && pathname !== "/dashboard") {
      const id = pathname.substring(1);
      const content = await env.GHOST_KV.get(id);
      
      if (!content) {
        return new Response(getNotFoundPage(FAVICON_URL), {
          status: 404,
          headers: { "Content-Type": "text/html; charset=utf-8", ...corsHeaders },
        });
      }

      // 阅后即焚核心
      await env.GHOST_KV.delete(id);

      return new Response(getViewPage(content, FAVICON_URL), {
        headers: { "Content-Type": "text/html; charset=utf-8", ...corsHeaders },
      });
    }

    return new Response(getNotFoundPage(FAVICON_URL), {
      status: 404,
      headers: { "Content-Type": "text/html; charset=utf-8", ...corsHeaders },
    });
  },
};

function escapeHtml(unsafe) {
  return unsafe
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}
