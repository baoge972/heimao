1生成 Base32 OTP 密钥

📍 在哪里运行：Windows PowerShell

按 Win + R，输入 powershell，按回车

方法 1：推荐（可复制）

$chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"
$key = -join (1..20 | ForEach-Object { Get-Random -InputObject $chars.ToCharArray() })
Write-Host "Base32 密钥: $key"

方法 2：一行版

-join (1..20 | ForEach-Object { "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[(Get-Random -Minimum 0 -Maximum 32)] })
✅ 输出示例：Base32 密钥: JBSWY3DPEHPK3PXP

密钥长度为 16-20 位，仅包含大写字母 A-Z 和数字 2-7

⚠️ 重要：必须使用 Base32 格式（仅 A-Z 和 2-7），不能使用 Base64！否则 Google Authenticator 会提示“密码值太短”

2配置 Workers 环境变量

  1. 登录 Cloudflare 控制台
  2. 进入 Workers & Pages → 选择你的 Worker
  3. 点击 设置 → 变量
  4. 在 环境变量 区域点击 添加变量
  5. 填写：
    • 变量名：OTP_SECRET
    • 变量值：步骤 1 生成的密钥（如 JBSWY3DPEHPK3PXP）
    • 类型：选择 密钥（加密存储）
  6. 点击 保存
  7. 部署 Worker 使环境变量生效
✅ 提示：如果不设置 OTP_SECRET，系统会自动跳过 OTP 验证（兼容旧版本）

3添加 TOTP 验证函数

将以下代码添加到 Workers 文件的顶部（路由定义之前）：

// --- TOTP 验证函数 ---
async function verifyTOTP(secret, token) {
  if (!secret || !token) return false;
  
  try {
    // Base32 解码
    function base32Decode(base32) {
      const base32Chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
      let bits = '';
      for (let i = 0; i < base32.length; i++) {
        const val = base32Chars.indexOf(base32[i].toUpperCase());
        if (val === -1) continue;
        bits += val.toString(2).padStart(5, '0');
      }
      
      const bytes = [];
      for (let i = 0; i + 8 <= bits.length; i += 8) {
        bytes.push(parseInt(bits.substring(i, i + 8), 2));
      }
      return new Uint8Array(bytes);
    }
    
    const key = base32Decode(secret.replace(/=/g, ''));
    
    // 当前时间步（30秒）
    const currentTimeStep = Math.floor(Date.now() / 30000);
    
    // 允许 ±2 时间步误差（±1分钟）
    for (let i = -2; i <= 2; i++) {
      const timeStep = currentTimeStep + i;
      
      // 生成时间步的8字节大端序数组
      const timeBytes = new ArrayBuffer(8);
      const view = new DataView(timeBytes);
      view.setUint32(4, timeStep, false);
      
      // HMAC-SHA1
      const cryptoKey = await crypto.subtle.importKey('raw', key, { name: 'HMAC', hash: 'SHA-1' }, false, ['sign']);
      const signature = await crypto.subtle.sign('HMAC', cryptoKey, timeBytes);
      
      // 动态截断
      const hmacResult = new Uint8Array(signature);
      const offset = hmacResult[hmacResult.length - 1] & 0xf;
      const binary = ((hmacResult[offset] & 0x7f) << 24) |
                     ((hmacResult[offset + 1] & 0xff) << 16) |
                     ((hmacResult[offset + 2] & 0xff) << 8) |
                     (hmacResult[offset + 3] & 0xff);
      
      const otp = (binary % 1000000).toString().padStart(6, '0');
      
      if (token === otp) return true;
    }
    
    return false;
  } catch (e) {
    console.error('TOTP 验证失败:', e);
    return false;
  }
}

4修改登录 API

找到你的登录 API 路由（如 /api/login），修改为两步验证：

router.post('/api/login', async (request, env) => {
    const { password, otp, step } = await request.json();
    
    // CORS headers
    const corsHeaders = {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'POST, OPTIONS',
        'Access-Control-Allow-Headers': 'Content-Type'
    };
    
    // 第一步：验证密码
    if (step === 1) {
        if (password !== env.ADMIN_PASSWORD) {
            return new Response(JSON.stringify({ success: false, message: '密码错误', step: 1 }), { 
                status: 401, 
                headers: { 
                    'Content-Type': 'application/json',
                    ...corsHeaders
                } 
            });
        }
        
        // 密码正确，检查是否需要 OTP
        if (env.OTP_SECRET && env.OTP_SECRET !== '') {
            return new Response(JSON.stringify({ success: true, requireOtp: true, step: 2 }), {
                headers: {
                    'Content-Type': 'application/json',
                    ...corsHeaders
                }
            });
        }
        
        // 不需要 OTP，直接登录成功
        return new Response(JSON.stringify({ success: true, step: 3 }), {
            headers: {
                'Content-Type': 'application/json',
                'Set-Cookie': `auth_token=authenticated; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=${60 * 60 * 24 * 7}`,
                ...corsHeaders
            }
        });
    }
    
    // 第二步：验证 OTP
    if (step === 2) {
        if (!otp) {
            return new Response(JSON.stringify({ success: false, message: '请输入动态验证码', step: 2 }), { 
                status: 401, 
                headers: { 
                    'Content-Type': 'application/json',
                    ...corsHeaders
                } 
            });
        }
        
        const isValid = await verifyTOTP(env.OTP_SECRET, otp);
        if (!isValid) {
            return new Response(JSON.stringify({ success: false, message: '动态验证码错误', step: 2 }), { 
                status: 401, 
                headers: { 
                    'Content-Type': 'application/json',
                    ...corsHeaders
                } 
            });
        }
        
        // OTP 验证成功
        return new Response(JSON.stringify({ success: true, step: 3 }), {
            headers: {
                'Content-Type': 'application/json',
                'Set-Cookie': `auth_token=authenticated; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=${60 * 60 * 24 * 7}`,
                ...corsHeaders
            }
        });
    }
});
✅ CORS 跨域支持：所有 API 响应已添加 Access-Control-Allow-Origin: * 等跨域 headers，支持前端跨域调用

JavaScript 部分

let currentPassword = '';

// 第一步：验证密码
document.getElementById('login-button').addEventListener('click', async () => {
    const password = document.getElementById('password').value;
    if (!password) {
        document.querySelector('.error-message').textContent = '请输入密码';
        return;
    }
    
    currentPassword = password;
    
    const response = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ password, step: 1 })
    });
    
    const data = await response.json();
    
    if (data.success) {
        if (data.requireOtp) {
            // 显示 OTP 模态框
            document.getElementById('otp-modal').style.display = 'flex';
            document.getElementById('otp-input').focus();
        } else {
            // 不需要 OTP，直接刷新
            window.location.reload();
        }
    } else {
        document.querySelector('.error-message').textContent = data.message || '登录失败';
    }
});

// 第二步：验证 OTP
document.getElementById('otp-submit-button').addEventListener('click', async () => {
    const otp = document.getElementById('otp-input').value;
    if (!otp || otp.length !== 6) {
        document.getElementById('otp-error').textContent = '请输入6位数字';
        return;
    }
    
    const response = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ password: currentPassword, otp, step: 2 })
    });
    
    const data = await response.json();
    
    if (data.success) {
        window.location.reload();
    } else {
        document.getElementById('otp-error').textContent = data.message || '验证失败';
        document.getElementById('otp-input').value = '';
        document.getElementById('otp-input').focus();
    }
});

// OTP 输入框自动提交
document.getElementById('otp-input').addEventListener('input', (e) => {
    e.target.value = e.target.value.replace(/\D/g, '');
    if (e.target.value.length === 6) {
        document.getElementById('otp-submit-button').click();
    }
});

常见问题

Q1：如果用户手机时间不准确怎么办？

A：代码已内置 ±2 时间步容差（±1 分钟），手机时间偏差 1 分钟内仍可验证通过。

Q2：可以不启用 OTP 吗？

A：可以。如果不设置 OTP_SECRET 环境变量，系统会自动跳过 OTP 验证，仅验证密码。

Q3：密钥丢失怎么办？

A：在 Workers 环境变量中重新设置新密钥，所有用户需用新密钥重新绑定。

Q4：支持多个用户吗？

A：当前方案所有用户共享同一密钥。如需独立密钥，需扩展为每个用户存储独立 Secret。

Q5：兼容性如何？

A：支持所有主流验证器 App：Google Authenticator、Microsoft Authenticator、Authy 等。
