const SALT = 'piledger.v1';
const VER = 'v1.0-new';

// 这里已经为你清空了所有历史数据，新人一进来就是空账本
const SEED = {
  income: [],
  expenses: [],
  params:{elecDaily:1.7355, netDaily:0.3458333333333333},
  meta:{version:1,imported:'2026-11-10',source:'new'}
};

async function sha256hex(s) {
  const d = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s));
  return [...new Uint8Array(d)].map(b => b.toString(16).padStart(2, '0')).join('');
}
function json(obj, status) {
  return new Response(JSON.stringify(obj), {
    status: status || 200,
    headers: {'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store'}
  });
}
function getCookie(req, name) {
  const h = req.headers.get('Cookie') || '';
  for (const part of h.split(';')) {
    const i = part.indexOf('=');
    if (i > -1 && part.slice(0, i).trim() === name) return part.slice(i + 1).trim();
  }
  return '';
}
async function getAuthHash(env) {
  const v = await env.DATA.get('auth');
  if (!v) return null;
  let o;
  try { o = JSON.parse(v); } catch (e) { return null; }
  if (!o || typeof o !== 'object') return null;
  if (typeof o.hash === 'string' && o.hash) return {hash: o.hash, set: o.set};
  if (typeof o.pw === 'string' && o.pw) return {pw: o.pw, set: o.set};
  return null;
}
async function authHashOf(stored) {
  if (stored.hash) return stored.hash;
  return await sha256hex(SALT + ':' + stored.pw);
}
async function isAuthed(req, env) {
  const stored = await getAuthHash(env);
  if (!stored) return false;
  const c = getCookie(req, 'pl_auth');
  if (!c) return false;
  return c === await authHashOf(stored);
}
function authCookie(hash) {
  return 'pl_auth=' + hash + '; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=2592000';
}
function clearCookie() {
  return 'pl_auth=; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=0';
}
async function getLedger(env) {
  let raw = await env.DATA.get('ledger');
  if (!raw) {
    await env.DATA.put('ledger', JSON.stringify(SEED));
    return JSON.parse(JSON.stringify(SEED));
  }
  let o;
  try { o = JSON.parse(raw); } catch (e) {
    throw new Error('账本数据损坏：KV 里的 ledger 值不是合法 JSON');
  }
  if (!o || !Array.isArray(o.income) || !Array.isArray(o.expenses)) {
    throw new Error('账本数据结构不对：缺少收益/费用列表');
  }
  let dirty = false;
  if (!o.params || typeof o.params.elecDaily !== 'number' || typeof o.params.netDaily !== 'number') {
    o.params = o.params || {};
    if (typeof o.params.elecDaily !== 'number') o.params.elecDaily = 1.7355;
    if (typeof o.params.netDaily !== 'number') o.params.netDaily = 0.3458333333333333;
    dirty = true;
  }
  const n0 = o.expenses.length;
  o.expenses = o.expenses.filter(r => r.type !== '电费' && r.type !== '网费');
  if (o.expenses.length !== n0) dirty = true;
  if (dirty) await env.DATA.put('ledger', JSON.stringify(o));
  return o;
}
async function readBody(req) {
  try { return await req.json(); } catch (e) { return null; }
}
function validateLedger(b) {
  if (!b || !Array.isArray(b.income) || !Array.isArray(b.expenses)) return '数据格式不对';
  for (const r of b.income) {
    if (!r || typeof r.start !== 'string' || typeof r.end !== 'string') return '收益记录缺少起止日期';
    if (typeof r.coins !== 'number' || !isFinite(r.coins) || r.coins < 0) return '收益金额非法';
    if (typeof r.days !== 'number' || !isFinite(r.days) || r.days < 0) return '收益天数非法';
  }
  for (const r of b.expenses) {
    if (!r || typeof r.start !== 'string') return '费用记录缺少日期';
    if (typeof r.type !== 'string') return '费用记录缺少类型';
    if (typeof r.amount !== 'number' || !isFinite(r.amount) || r.amount < 0) return '费用金额非法';
    if (r.days != null && (typeof r.days !== 'number' || r.days < 0)) return '费用天数非法';
  }
  return null;
}

async function handle(request, env) {
  const url = new URL(request.url);
  const path = url.pathname;

  if (path === '/api/ping') return json({ok: true, ver: VER, time: new Date().toISOString()});
  if (path === '/api/state') {
    const stored = await getAuthHash(env);
    return json({initialized: !!stored, authed: await isAuthed(request, env), ver: VER});
  }
  if (path === '/api/setup' && request.method === 'POST') {
    const stored = await getAuthHash(env);
    if (stored) return json({ok: false, error: '口令已设置，请直接登录'}, 403);
    const b = await readBody(request);
    const pw = b && b.password ? String(b.password).trim() : '';
    if (pw.length < 4) return json({ok: false, error: '口令至少 4 位'}, 400);
    const hash = await sha256hex(SALT + ':' + pw);
    await env.DATA.put('auth', JSON.stringify({hash: hash, set: new Date().toISOString()}));
    const after = await getAuthHash(env);
    if (!after || (await authHashOf(after)) !== hash) return json({ok: false, error: '口令已被并发设置'}, 409);
    return new Response(JSON.stringify({ok: true}), {headers: {'Content-Type': 'application/json', 'Set-Cookie': authCookie(hash)}});
  }
  if (path === '/api/login' && request.method === 'POST') {
    const stored = await getAuthHash(env);
    if (!stored) return json({ok: false, error: '尚未设置口令'}, 400);
    const b = await readBody(request);
    const pw = b && b.password ? String(b.password) : '';
    const hash = await sha256hex(SALT + ':' + pw);
    if (hash !== await authHashOf(stored)) {
      await new Promise(r => setTimeout(r, 800));
      return json({ok: false, error: '口令错误'}, 401);
    }
    return new Response(JSON.stringify({ok: true}), {headers: {'Content-Type': 'application/json', 'Set-Cookie': authCookie(hash)}});
  }
  if (path === '/api/logout' && request.method === 'POST') {
    return new Response(JSON.stringify({ok: true}), {headers: {'Content-Type': 'application/json', 'Set-Cookie': clearCookie()}});
  }
  if (path === '/api/password' && request.method === 'POST') {
    if (!(await isAuthed(request, env))) return json({ok: false, error: '未登录'}, 401);
    const b = await readBody(request);
    const oldPw = b && b.old ? String(b.old) : '';
    const newPw = b && b.new ? String(b.new).trim() : '';
    const stored = await getAuthHash(env);
    if (!stored) return json({ok: false, error: '尚未设置口令'}, 400);
    const oldHash = await sha256hex(SALT + ':' + oldPw);
    if (oldHash !== await authHashOf(stored)) return json({ok: false, error: '原口令错误'}, 401);
    if (newPw.length < 4) return json({ok: false, error: '新口令至少 4 位'}, 400);
    const newHash = await sha256hex(SALT + ':' + newPw);
    await env.DATA.put('auth', JSON.stringify({hash: newHash, set: new Date().toISOString()}));
    return new Response(JSON.stringify({ok: true}), {headers: {'Content-Type': 'application/json', 'Set-Cookie': authCookie(newHash)}});
  }
  if (path === '/api/data') {
    if (!(await isAuthed(request, env))) return json({ok: false, error: '未登录'}, 401);
    if (request.method === 'GET') return json({ok: true, data: await getLedger(env)});
    if (request.method === 'POST') {
      const b = await readBody(request);
      const err = validateLedger(b);
      if (err) return json({ok: false, error: err}, 400);
      if (JSON.stringify(b).length > 500000) return json({ok: false, error: '数据太大'}, 400);
      var prm = (b.params && typeof b.params.elecDaily === 'number' && typeof b.params.netDaily === 'number')
        ? {elecDaily: +b.params.elecDaily, netDaily: +b.params.netDaily}
        : {elecDaily: 1.7355, netDaily: 0.3458333333333333};
      await env.DATA.put('ledger', JSON.stringify({income: b.income, expenses: b.expenses, params: prm, meta: b.meta || {version: 1}}));
      return json({ok: true});
    }
  }
  if (path === '/' || path === '/index.html') {
    return new Response(HTML, {headers: {'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store'}});
  }
  return new Response('Not Found', {status: 404});
}

export default {
  async fetch(request, env) {
    try { return await handle(request, env); } 
    catch (e) { return json({ok: false, error: '后端出错：' + (e && e.message ? e.message : String(e))}, 500); }
  }
};

function barChart(items){
  if(!items.length)return '<div class="empty">暂无数据</div>';
  var w=Math.max(320,items.length*34),h=178,pad=26;
  var CH=h-pad-16;
  var TOP=20;
  
  var max=0;items.forEach(function(i){if(i.v>max)max=i.v;});
  if(max<=0)max=1;
  var bw=w/items.length;
  var bars=items.map(function(it,i){
    var bh=(CH-TOP)*(it.v/max);
    var x=i*bw+bw*0.15, y=CH-bh;
    
    var r='<rect class="bar" x="'+x.toFixed(1)+'" y="'+y.toFixed(1)+'" width="'+(bw*0.7).toFixed(1)+'" height="'+Math.max(bh,0.5).toFixed(1)+'" rx="2"><title>'+esc(it.label)+'：'+fmt(it.v,2)+'</title></rect>';
    r += '<text class="axlabel" x="'+(x+bw*0.35).toFixed(1)+'" y="'+(y-4).toFixed(1)+'" text-anchor="middle" font-size="11">'+(it.v>=100?Math.round(it.v):it.v.toFixed(1))+'</text>';
    r += '<text class="axlabel" x="'+(x+bw*0.35).toFixed(1)+'" y="'+(h-pad+2)+'" text-anchor="middle" font-size="11">'+esc(it.label)+'</text>';
    return r;
  }).join('');
  
  return '<div class="chart"><svg width="'+w+'" height="'+h+'" viewBox="0 0 '+w+' '+h+'">'
    + '<line class="axis" x1="0" y1="'+(h-pad-16)+'" x2="'+w+'" y2="'+(h-pad-16)+'"/>' 
    + bars 
    + '</svg></div>';
}

function bindForms(){
  var bx=$('btnXls'),bd=$('btnDoc');
  if(bx)bx.onclick=exportXls;
  if(bd)bd.onclick=exportDoc;
  
  var ps=$('pSave');
  if(ps)ps.onclick=function(){
    var e=parseFloat($('pElec').value),n=parseFloat($('pNet').value);
    if(isNaN(e)||e<0||isNaN(n)||n<0)return alert('单价要是非负数字');
    S.data.params={elecDaily:e,netDaily:n};
    save();
  };

  var sv=$('fSave');
  if(sv)sv.onclick=function(){
    var start=$('fStart').value,end=$('fEnd').value,coins=parseFloat($('fCoins').value),note=$('fNote').value.trim();
    if(!start||!end)return alert('请填开始和结束日期');
    var days=daysInclusive(start,end);
    if(days==null||days<1)return alert('结束日期要晚于开始日期');
    if(!(coins>=0)||isNaN(coins))return alert('请填收益数量');
    
    if(S.editingIncome){
      var r=S.editingIncome;
      var isSeed = r.id.indexOf('i')===0;
      if(!isSeed){
        r.start=start;
        r.end=end;
        r.days=days;
        r.note=note;
      }
      r.coins=coins;
      S.editingIncome=null;
    }else{
      S.data.income.push({id:uid(),start:start,end:end,days:days,coins:coins,note:note});
    }
    save();
  };
  var fc=$('fCancel');
  if(fc)fc.onclick=function(){S.editingIncome=null;renderMain();};

  var esv=$('fESave');
  if(esv)esv.onclick=function(){
    var type=$('fType').value,amount=parseFloat($('fAmount').value),start=$('fEStart').value,end=$('fEEnd').value,note=$('fENote').value.trim();
    if(!start)return alert('请填日期');
    if(!(amount>=0)||isNaN(amount))return alert('请填金额');
    var rec={type:type,amount:amount,start:start,end:end||start,note:note};
    if(type==='云服'&&end&&end!==start){
      var d=daysInclusive(start,end);
      if(d==null||d<1)return alert('云服结束日期要晚于开始日期');
      rec.days=d;
    }
    if(S.editingExpense){
      var r=S.editingExpense;
      r.type=rec.type;r.amount=rec.amount;r.start=rec.start;r.end=rec.end;r.note=rec.note;
      if(rec.days)r.days=rec.days;else delete r.days;
      S.editingExpense=null;
    }else{
      rec.id=uid();
      S.data.expenses.push(rec);
    }
    save();
  };
  var fec=$('fECancel');
  if(fec)fec.onclick=function(){S.editingExpense=null;renderMain();};
  
  document.querySelectorAll('[data-ei]').forEach(function(b){b.onclick=function(){editIncome(b.getAttribute('data-ei'));};});
  document.querySelectorAll('[data-di]').forEach(function(b){b.onclick=function(){delIncome(b.getAttribute('data-di'));};});
  document.querySelectorAll('[data-ee]').forEach(function(b){b.onclick=function(){editExpense(b.getAttribute('data-ee'));};});
  document.querySelectorAll('[data-de]').forEach(function(b){b.onclick=function(){delExpense(b.getAttribute('data-de'));};});
}

function exportXls(){
  ensureXlsx(function(){
    var c=calc();
    var wb=XLSX.utils.book_new();
    var sum=[
      ['Pi 节点记账 · 汇总'],
      [],
      ['累计收益（个）',c.totalCoins],['运行核算（天）',c.totalDays],['日均收益（个/天）',+c.dailyAvg.toFixed(4)],
      ['累计投入（元）',+c.totalCost.toFixed(2)],
      ['单币成本·全成本（元/个）',+c.fullUnit.toFixed(4)],
      ['单币成本·仅运营（元/个）',+c.opUnit.toFixed(4)],
      []
    ];
    Object.keys(c.years).sort().forEach(function(y){
      sum.push([y+'年 收益（个）',c.years[y].coins],[y+'年 核算天数',c.years[y].days]);
    });
    sum.push([]);sum.push(['费用构成']);
    Object.keys(c.byType).forEach(function(t){sum.push([t+'（元）',+c.byType[t].toFixed(2)]);});
    sum.push(['电费单价（元/天）',+c.elecDaily]);
    sum.push(['网费单价（元/天）',+c.netDaily]);
    sum.push(['电费/网费说明','按「单价 × 运行核算天数」折算，未逐笔计']);
    XLSX.utils.book_append_sheet(wb,XLSX.utils.aoa_to_sheet(sum),'汇总');
    var inc=[['开始日期','结束日期','核算天数','收益数量（个）','期间日均','备注']];
    S.data.income.slice().sort(function(a,b){return a.start<b.start?-1:1}).forEach(function(r){
      inc.push([r.start,r.end,+r.days,+r.coins,+((+r.days?r.coins/r.days:0).toFixed(4)),r.note||'']);
    });
    XLSX.utils.book_append_sheet(wb,XLSX.utils.aoa_to_sheet(inc),'收益明细');
    var exp=[['类型','开始日期','结束日期','云服天数','金额（元）','备注']];
    S.data.expenses.slice().sort(function(a,b){return a.start<b.start?-1:1}).forEach(function(r){
      exp.push([r.type,r.start,r.end,r.days||'',+r.amount,r.note||'']);
    });
    XLSX.utils.book_append_sheet(wb,XLSX.utils.aoa_to_sheet(exp),'费用明细');
    XLSX.writeFile(wb,'Pi节点记账-'+today()+'.xlsx');
  });
}
